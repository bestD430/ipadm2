// 搜尋路線與方向 (修正城巴 API 域名與路徑) function fetchDirections(idx) { var company = document.getElementById('company-' + idx).value; var route = document.getElementById('route-input-' + idx).value.trim().toUpperCase(); if (!route) return; busConfigs[idx].company = company; busConfigs[idx].route = route; var dirSelect = document.getElementById('dir-' + idx); var stopSelect = document.getElementById('stop-' + idx); dirSelect.innerHTML = ''; stopSelect.innerHTML = ''; stopSelect.disabled = true; if (company === 'KMB') { fetch('https://data.etabus.gov.hk/v1/transport/kmb/route/') .then(function(res) { return res.json(); }) .then(function(data) { dirSelect.innerHTML = ''; if (data.data) { var matched = data.data.filter(function(r) { return r.route === route; }); if (matched.length > 0) { matched.forEach(function(item) { var opt = document.createElement('option'); var boundCode = item.bound === 'O' ? 'outbound' : 'inbound'; opt.value = boundCode + '|' + item.service_type; opt.innerText = '往：' + item.dest_tc; dirSelect.appendChild(opt); }); dirSelect.disabled = false; } else { dirSelect.innerHTML = ''; } } }); } else { // 修正：使用 rt.data.gov.hk/v2/ 專屬城巴 API fetch('https://rt.data.gov.hk/v2/transport/citybus/route/ctb/' + route) .then(function(res) { return res.json(); }) .then(function(res) { dirSelect.innerHTML = ''; var routeData = res.data; if (routeData && routeData.dest_tc) { busConfigs[idx].ctbCo = 'ctb'; var optOut = document.createElement('option'); optOut.value = 'outbound|1'; optOut.innerText = '往：' + routeData.dest_tc; dirSelect.appendChild(optOut); var optIn = document.createElement('option'); optIn.value = 'inbound|1'; optIn.innerText = '往：' + routeData.orig_tc; dirSelect.appendChild(optIn); dirSelect.disabled = false; } else { dirSelect.innerHTML = ''; } }) .catch(function() { dirSelect.innerHTML = ''; }); } } // 搜尋車站 (修正城巴 API 網址) function fetchStops(idx) { var dirVal = document.getElementById('dir-' + idx).value; if (!dirVal) return; var parts = dirVal.split('|'); busConfigs[idx].bound = parts[0]; busConfigs[idx].serviceType = parts[1]; var stopSelect = document.getElementById('stop-' + idx); stopSelect.innerHTML = ''; var url = busConfigs[idx].company === 'KMB' ? 'https://data.etabus.gov.hk/v1/transport/kmb/route-stop/' + busConfigs[idx].route + '/' + busConfigs[idx].bound + '/' + busConfigs[idx].serviceType : 'https://rt.data.gov.hk/v2/transport/citybus/route-stop/ctb/' + busConfigs[idx].route + '/' + busConfigs[idx].bound; fetch(url) .then(function(res) { return res.json(); }) .then(function(data) { stopSelect.innerHTML = ''; if (data.data && data.data.length > 0) { var promises = data.data.slice(0, 35).map(function(stop) { var stopUrl = busConfigs[idx].company === 'KMB' ? 'https://data.etabus.gov.hk/v1/transport/kmb/stop/' + stop.stop : 'https://rt.data.gov.hk/v2/transport/citybus/stop/' + stop.stop; return fetch(stopUrl).then(function(r) { return r.json(); }); }); Promise.all(promises).then(function(stopDetails) { stopDetails.forEach(function(detail, index) { if (detail.data) { var opt = document.createElement('option'); opt.value = detail.data.stop; opt.innerText = (index + 1) + '. ' + detail.data.name_tc; stopSelect.appendChild(opt); } }); stopSelect.disabled = false; }); } }); } // 取得即時到站時間 (修正城巴 API 網址) function fetchETAForIndex(idx) { var cfg = busConfigs[idx]; if (!cfg.stopId || !cfg.active) return; var url = cfg.company === 'KMB' ? 'https://data.etabus.gov.hk/v1/transport/kmb/eta/' + cfg.stopId + '/' + cfg.route + '/' + cfg.serviceType : 'https://rt.data.gov.hk/v2/transport/citybus/eta/ctb/' + cfg.stopId + '/' + cfg.route; fetch(url) .then(function(res) { return res.json(); }) .then(function(data) { var etaDisplay = document.getElementById('eta-' + idx); if (data.data && data.data.length > 0) { var validEtas = data.data .filter(function(item) { return item.eta !== null && item.eta !== ''; }) .slice(0, 3) .map(function(item) { var diff = Math.round((new Date(item.eta) - new Date()) / 60000); return diff <= 0 ? '即將到站' : diff + ' 分鐘'; }); if (validEtas.length === 0) { etaDisplay.innerHTML = '
尾班車已過或暫無班次
'; return; } var html = '
' + validEtas[0] + '
'; if (validEtas.length > 1) { html += '
'; if (validEtas[1]) html += '
下班: ' + validEtas[1] + '
'; if (validEtas[2]) html += '
下下班: ' + validEtas[2] + '
'; html += '
'; } etaDisplay.innerHTML = html; } else { etaDisplay.innerHTML = '
暫無即時到站資料
'; } }); }
